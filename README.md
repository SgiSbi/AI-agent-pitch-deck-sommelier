# Наш командный проект
Здесь будем собирать и структурировать файлы.

Ai-agent/     
│
├── data/                     # Исходные данные и промежуточные результаты   
│   ├── raw/                  # Оригинальные презентации (.pptx, .pdf)    
│   ├── slides/               # Слайды в виде изображений (после разбивки)    
│   └── annotations/          # Ручные/автоматические описания слайдов (JSON, TXT, CSV)    
│     
├── prompts/                  # Промпты для LLM     
|   ├── slyde_reader/         # Промпты для чтения презентаций    
│   ├── block_analysis/       # Промпты для анализа блоков    
│   ├── report_generation/    # Промпты для генерации финального отчёта    
│   └── evaluation/           # Промпты для оценки качества    
│     
├── reports/                  # Готовые и черновые отчёты    
│   ├── drafts/               # Черновики, эксперименты    
│   └── final/                # Финальные версии отчётов    
│    
├── examples/                 # Примеры "как должно быть"    
│   ├── input_presentation/   # Примеры входных презентаций    
│   └── expected_output/      # Ожидаемые отчёты / структуры    
│    
├── docs/                     # Документация проекта    
│    
├── scripts/                  # Вспомогательные скрипты    
│    
├── .gitignore                # Игнорирование временных/системных файлов    
├── README.md                 # Описание проекта, как начать, структура    
└── requirements.txt          # (Опционально) зависимости Python
